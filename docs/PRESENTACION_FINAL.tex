\begin{frame}{De la propuesta al informe final}

\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Pregunta de investigación}:
\begin{quote}\small\itshape
¿Produce la CWT-Morlet un espectrograma de modulación con mayor poder
discriminativo para AD que la STFT?
\end{quote}

\vspace{4pt}
\textbf{Lo que se hizo desde el avance:}
\begin{itemize}\small
    \item Multi-seed (3~seeds) sobre las 6~configuraciones.
    \item DeLong por seed + Stouffer/Fisher (inferencia correcta).
    \item BH-FDR sobre las 3~comparaciones principales.
    \item Post-experimentos: saliency, bandas, canales, género.
    \item Auditoría externa en dos rondas.
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Hipótesis original}
\small CWT $>$ STFT por mejor resolución frecuencial en bajas Hz.
\end{block}
\begin{alertblock}{Resultado final}
\small \textbf{No confirmada}, pero la dirección opuesta tampoco se prueba
con rigor estadístico.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Confound DSP: el eje de modulación no es comparable}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Ambas imágenes finales son $45 \times 45$, pero NO codifican lo mismo:}

\begin{table}
\centering
\scriptsize
\begin{tabular}{l c c}
\toprule
 & STFT & CWT \\
\midrule
$dt$ envolvente   & $0.32$ s & $0.005$ s \\
Nyquist mod. real & $\mathbf{1.5625}$ Hz & $100$ Hz \\
Bins en $[0,22.5]$ Hz & $13$ & $\sim 180$ \\
Estrategia para 45 bins & \emph{interpolación} & \emph{subsample} \\
Rango efectivo & $[0, 1.56]$ Hz & $[0, 22.5]$ Hz \\
\bottomrule
\end{tabular}
\end{table}

\vfill
\begin{alertblock}{Implicación}
\small
La comparación STFT vs CWT en este pipeline \textbf{NO es una
comparación justa entre transformadas T-F}: es entre dos pipelines con
ejes de modulación drásticamente distintos.
\end{alertblock}

% SLIDE 10: Limitaciones honestas
% ============================================================
\begin{frame}{Limitaciones honestas (16 documentadas en el informe)}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Críticas (afectan la conclusión principal)}:
\begin{itemize}\small
    \item \textbf{Confound DSP del eje de modulación} (STFT
    Nyquist $1.56$ Hz vs CWT $100$ Hz). La limitación más seria.
    \item \textbf{Solo 3 seeds} (poder estadístico modesto).
    \item \textbf{CWT subexplorada}: solo \texttt{cmor1.5-1.0},
    32~escalas log; sin tuning específico.
    \item \textbf{Resize bilinear $45 \times 45$} oculta el desajuste
    anterior.
\end{itemize}

\vfill
\textbf{Metodológicas}:
\begin{itemize}\small
    \item Grid search de patches heurístico (no nested CV).
    \item BH-FDR post-hoc, no pre-registrado.
    \item Sin validación externa (otro dataset).
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{De interpretabilidad}:
\begin{itemize}\small
    \item Saliency \emph{unstable} fold-to-fold (Jaccard $\approx 0.05$).
    \item Análisis post-hoc (bandas, canales, género) no
    pre-registrados.
    \item Atribución por banda restringida a STFT (eje CWT geomspace).
\end{itemize}

\vfill
\textbf{Sobre el dataset}:
\begin{itemize}\small
    \item Sesgo de género ($\chi^2$ $p = 0.039$), Fisher F vs M
    $p = 0.201$ pero poder $\sim 25$--$30\%$
    ($N \approx 64$/grupo requerido).
    \item \emph{Ausencia de evidencia $\neq$ evidencia de ausencia}.
    \item HC con MMSE perfecto (30/30) --- posible \emph{cherry-picking}
    upstream.
\end{itemize}
\end{column}
\end{columns}

\end{frame}

% SLIDE 11: Conclusiones
% ============================================================
\begin{frame}{Conclusiones}

\begin{enumerate}
    \item \textbf{Replicación funcional exitosa} del pipeline de Lopes
    sobre dataset público \texttt{ds004504}: SVM vainilla + STFT
    alcanza Acc $0.764 \pm 0.009$, AUC $0.856 \pm 0.022$, en el rango
    del paper original.

    \item \textbf{La comparación STFT vs CWT no es concluyente}:
    \begin{itemize}\small
        \item DeLong por seed + Stouffer: $p \approx 0.061$ (NS).
        \item Tras BH-FDR: \textbf{ninguna comparación sobrevive}
        ($q \geq 0.10$).
        \item Adicionalmente, los ejes de modulación entre STFT y CWT
        son \textbf{físicamente distintos} --- la comparación está
        confundida por DSP, no controlada por diseño.
    \end{itemize}

    \item \textbf{El método de saliency cambia el ranking}: con
    Grad-CAM, CWT $>$ STFT (NS); con vainilla, STFT $>$ CWT (NS). Las
    saliency maps son \textbf{estadísticamente ortogonales}
    $\Rightarrow$ \emph{posible complementariedad} (ensemble como
    hipótesis futura).

    \item \textbf{Observación clínicamente coherente (post-hoc)}: STFT
    vainilla concentra saliency en $\alpha + \theta$ y canales
    occipito-temporales --- convergente con literatura clínica EEG-AD.

    \item \textbf{Aporte metodológico para la comunidad}: reportar
    siempre varianza por seed, método de saliency, y poder estadístico
    al afirmar ``ausencia de efecto''.
\end{enumerate}

\end{frame}
