\begin{frame}{La pregunta}

\begin{columns}[T]
\begin{column}{0.57\textwidth}
\small
\begin{itemize}
    \item El Alzheimer es la demencia \#1; detectarlo temprano importa,
    pero MRI/PET son caros.
    \item El \textbf{EEG} es barato y portátil; su \textbf{espectro de
    modulación} captura \emph{a qué ritmo} cambia la energía de cada banda.
    \item Todo depende del primer paso: la \textbf{descomposición
    tiempo-frecuencia}.
\end{itemize}

\begin{block}{Hipótesis}
\footnotesize La \textbf{CWT}, con mejor resolución en bajas frecuencias,
debería superar a la \textbf{STFT} para el Alzheimer.
\end{block}
\end{column}

% SLIDE 3a: Zoom DSP (1/3) — limpieza
% ============================================================
\begin{frame}{Zoom DSP (1/3) --- Limpieza: FIR 0.5--45 Hz + ICA}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\small\textbf{Qué hago}
\begin{itemize}
    \item \textbf{FIR pasa-banda 0.5--45 Hz} (fase lineal).
    \item \textbf{ICA + ICLabel} (+ wICA en residuos): separa y descarta
    componentes \emph{no} cerebrales (ojo, músculo, línea, ECG).
\end{itemize}
\end{column}
\begin{column}{0.52\textwidth}
\small\textbf{Por qué (DSP)}
\begin{itemize}
    \item \textbf{0.5 Hz} (corte alto): quita deriva/DC y sudor, por
    debajo del EEG útil --- conserva delta.
    \item \textbf{45 Hz} (corte bajo): elimina línea (50/60 Hz) y EMG.
    \item \textbf{FIR, no IIR}: \emph{fase lineal} $\Rightarrow$ no
    distorsiona el \emph{tiempo} de la envolvente --- y esa envolvente es
    justo lo que mide el espectro de modulación.
    \item \textbf{ICLabel automático}: objetivo y reproducible en 65
    sujetos.
\end{itemize}
\end{column}
\end{columns}

\vfill
\begin{alertblock}{Si no}
\small Un parpadeo o el EMG son \textbf{transitorios de alta potencia}: sin
quitarlos, el espectrograma podría confundirlos con un biomarcador.
\end{alertblock}

\end{frame}

% ============================================================
% SLIDE 3b: Zoom DSP (2/3) — acondicionamiento
% ============================================================
\begin{frame}{Zoom DSP (2/3) --- Acondicionamiento: 200 Hz y épocas de 8 s}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\small\textbf{Qué hago}
\begin{itemize}
    \item \textbf{Remuestreo 500 $\to$ 200 Hz} (con anti-aliasing).
    \item \textbf{Épocas de 8 s} con solape de 1 s.
\end{itemize}
\end{column}
\begin{column}{0.52\textwidth}
\small\textbf{Por qué (DSP)}
\begin{itemize}
    \item Tras filtrar a 45 Hz basta Nyquist $>90$ Hz: \textbf{200 Hz}
    (Nyquist 100) sobra y evita aliasing, con 2.5$\times$ menos datos.
    \item \textbf{8 s} $\Rightarrow$ resolución de modulación
    $\Delta f_{\text{mod}} = 1/T = \mathbf{0.125}$ Hz: fina para las
    modulaciones \textbf{lentas} ($<2$ Hz) relevantes en AD.
    \item Solape 1 s $\Rightarrow$ más épocas (\emph{augmentation}).
\end{itemize}
\end{column}
\end{columns}

\vfill
\begin{block}{El compromiso}
\small Época \textbf{más corta} = peor resolución de modulación; \textbf{más
larga} = se rompe la cuasi-estacionariedad del EEG. 8 s es el punto medio
(y el del paper).
\end{block}

\end{frame}

% ============================================================
% SLIDE 3c: Zoom DSP (3/3) — representación T-F + modspec
% ============================================================
\begin{frame}{Zoom DSP (3/3) --- Representación T-F + espectro de modulación}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\small\textbf{Qué hago}
\begin{itemize}
    \item \textbf{STFT \emph{o} CWT-Morlet} $\to X(t,f)$:
    \begin{itemize}\scriptsize
        \item STFT: ventana \emph{fija} $\Rightarrow \Delta t\cdot\Delta f$
        constante.
        \item CWT: ventana \emph{por escala} $\Rightarrow$ mejor $\Delta f$
        en bajas frecuencias.
    \end{itemize}
    \item \textbf{Modspec}: $|X|^2 \to \mathcal{F}_t \to
    M(f,f_{\text{mod}})$, redimensionado a \textbf{45$\times$45} (entrada
    de la CNN).
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\small\textbf{Por qué (DSP)}
\begin{itemize}
    \item STFT vs CWT es \textbf{la variable en estudio} (hipótesis: CWT
    mejor en bajas $\Rightarrow$ AD).
    \item \textbf{45$\times$45}: tamaño de entrada fijo que exige la CNN
    (fiel al paper).
\end{itemize}
\vspace{3pt}
\begin{alertblock}{Sutileza (la retomo en ``el giro'')}
\scriptsize Forzar ambos a 45$\times$45 \textbf{oculta} que el eje de
modulación no mide lo mismo en STFT y CWT: ese es el \emph{confound} que
controlo con CWT-fair.
\end{alertblock}
\end{column}
\end{columns}

\end{frame}

% ============================================================
% SLIDE 5: El giro (confound identificado y controlado)
% ============================================================
\begin{frame}{El giro: no estaba comparando peras con peras}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\small
Las dos imágenes son $45\times45$, pero su \textbf{eje de modulación}
(vertical) no mide lo mismo:

\begin{table}
\centering
\scriptsize
\begin{tabular}{l c c}
\toprule
 & STFT & CWT nativa \\
\midrule
Nyquist modulación & $\mathbf{1.56}$ Hz & $100$ Hz \\
Bins reales & $\sim$13 & $\sim$180 \\
\bottomrule
\end{tabular}
\end{table}

\vfill
Una comparación STFT vs CWT mezcla \textbf{dos cosas}: la transformada
\emph{y} la resolución del eje de modulación.
\end{column}
\begin{column}{0.47\textwidth}
\begin{alertblock}{La solución: CWT-fair}
\small Una tercera condición: la \textbf{misma CWT}, pero con la
envolvente decimada para que su eje de modulación \textbf{iguale al de
la STFT}.
\end{alertblock}

\small
\begin{itemize}
    \item \textbf{STFT vs CWT-fair} $\Rightarrow$ aísla la
    \emph{transformada} (comparación justa).
    \item \textbf{CWT nativa vs CWT-fair} $\Rightarrow$ mide el efecto
    del \emph{eje} (el artefacto).
\end{itemize}
\end{column}
\end{columns}

\end{frame}

% SLIDE 6: El resultado (convergencia)
% ============================================================
\begin{frame}{El resultado: al igualar el eje, STFT $\approx$ CWT}

\begin{columns}[T]
\begin{column}{0.54\textwidth}
\centering
\small AUC (3 seeds) --- CWT-fair \textbf{se mueve hacia STFT} en las dos direcciones:
\begin{table}
\centering
\small
\begin{tabular}{l c c c}
\toprule
Saliency & STFT & CWT nativa & \textbf{CWT-fair} \\
\midrule
Vainilla & $0.856$ & $0.778$ & $\mathbf{0.828}$ \\
Grad-CAM & $0.713$ & $0.800$ & $\mathbf{0.769}$ \\
\bottomrule
\end{tabular}
\end{table}
\vspace{2pt}
{\scriptsize Vainilla: brecha $-0.078 \to -0.028$ ($\sim$64\%). \quad
Grad-CAM: $+0.087 \to +0.056$ ($\sim$36\%).}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Comparación justa (STFT vs CWT-fair)}
\small DeLong por seed + Stouffer:
\begin{itemize}
    \item Vainilla: $p = 0.318$
    \item Grad-CAM: $p = 0.587$
\end{itemize}
\centering\footnotesize Ninguna sobrevive BH-FDR ($q \approx 0.68$).
\end{block}
\end{column}
\end{columns}

\vfill
\centering
\textbf{A igualdad de eje de modulación, STFT y CWT-Morlet son
indistinguibles.} \\
\small La diferencia aparente (en ambos sentidos) era \textbf{en gran
parte} un artefacto de DSP, no la transformada.
{\scriptsize\ (CWT-fair iguala ``hacia abajo'': descarta modulaciones
$>1.56$ Hz; es 1 de 2 definiciones de comparación justa.)}

\end{frame}

% SLIDE 7: Lo que me llevo + cierre

\begin{frame}{Lo que me llevo}

\begin{columns}[T]
\begin{column}{0.60\textwidth}
\begin{enumerate}
    \item \textbf{La replicación funciona} sobre datos públicos ---
    el método de Lopes es reproducible (AUC $0.856$).
    \item \textbf{Comparar transformadas T-F exige igualar el eje de
    modulación.} Sin ese control, un artefacto de DSP puede invertir el
    ranking aparente. Con el control (CWT-fair), STFT y CWT empatan.
    \item \textbf{Aporte metodológico} $>$ el resultado numérico:
    identificar el confound, diseñar el control, y demostrar que la
    diferencia era un artefacto.
\end{enumerate}

\vfill
\small\textbf{A futuro}: igualar el eje ``hacia arriba'' (STFT con hop
fino) para ver si las modulaciones rápidas de la CWT aportan algo.
\end{column}
